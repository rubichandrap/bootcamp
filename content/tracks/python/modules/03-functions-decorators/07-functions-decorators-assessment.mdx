  from solution import compose, make_taxer, repeat, greet, say_hello


  def test_compose_applies_g_then_f():
      h = compose(lambda x: x * 2, lambda x: x + 1)
      assert h(3) == 8


  def test_compose_named_functions():
      def double(x):
          return x * 2

      def add_one(x):
          return x + 1

      h = compose(double, add_one)
      assert h(5) == 12


  def test_make_taxer_twenty_five_percent():
      tax = make_taxer(0.25)
      assert tax(100) == 125.0


  def test_make_taxer_no_tax():
      tax = make_taxer(0)
      assert tax(50) == 50.0


  def test_repeat_calls_three_times():
      calls = []

      @repeat(3)
      def record(value):
          calls.append(value)
          return value

      record("hi")
      assert calls == ["hi", "hi", "hi"]


  def test_repeat_returns_last_result():
      total = 0

      @repeat(3)
      def add_one():
          nonlocal total
          total += 1
          return total

      assert add_one() == 3


  def test_repeat_once():
      @repeat(1)
      def square(x):
          return x * x

      assert square(4) == 16


  def test_say_hello_partial():
      assert say_hello("Ada") == "Hello, Ada!"

      assert greet("Hi", "Bob") == "Hi, Bob!"
