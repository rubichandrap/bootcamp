  import { describe, it, expect } from 'vitest';
  import { withTimeout, settleAll, firstSuccess, batchedAll } from './solution';

  describe('withTimeout', () => {
    it('resolves if promise completes in time', async () => {
      const result = await withTimeout(Promise.resolve(42), 100);
      expect(result).toBe(42);
    });

    it('rejects with timeout error if too slow', async () => {
      const slow = new Promise<number>((resolve) => setTimeout(() => resolve(1), 200));
      await expect(withTimeout(slow, 50)).rejects.toThrow();
    });
